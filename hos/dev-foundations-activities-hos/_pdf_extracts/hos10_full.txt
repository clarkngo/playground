5. Copy the following code into your “lambda_function.py” file. 
 
import json  
 
def lambda_handler(event, context):  
 
    query_params = event['queryStringParameters']  
    query_params = int(query_params['year'])  
    result = str(query_params) + (' is a leap year' if is_leap_year(int(query_params)) else ' is not a leap 
year')  
    return {  
        'statusCode': 200,  
        'body': json.dumps(result)  
    }  
 
def is_leap_year(year):  
    if year % 4 == 0:  
        if year % 100 == 0:  
            if year % 400 == 0:  
                return True  
            else:  
                return False  
        else:  
            return True  
    else:  
        return False
