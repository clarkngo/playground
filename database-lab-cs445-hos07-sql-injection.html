def init_db():
    # Generates demo.db [cite: 40]
    conn = sqlite3.connect('demo.db')
    cursor = conn.cursor()
    cursor.execute('CREATE TABLE users (id INT, username TEXT, password TEXT)')
    # Normal credentials [cite: 45, 46]
    cursor.execute("INSERT INTO users VALUES (1, 'alice', 'alicepass')")
    cursor.execute("INSERT INTO users VALUES (2, 'admin', 'hidden_admin_pass')")
    conn.commit()
    conn.close()

import sqlite3

# Secure login using parameters [cite: 50]
def login(username, password):
    conn = sqlite3.connect('demo.db')
    cursor = conn.cursor()
    # SECURE: '?' placeholders block malicious actors [cite: 23]
    query = "SELECT * FROM users WHERE username = ? AND password = ?"
    cursor.execute(query, (username, password))
    return cursor.fetchone()
