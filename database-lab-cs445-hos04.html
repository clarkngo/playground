1. CREATE Table

Creates the schema for employee records[cite: 291].

CREATE TABLE employee ( id int NOT NULL PRIMARY KEY, name varchar(255) NOT NULL, email varchar(255), gender varchar(255), designation varchar(255) );

2. INSERT Data

Populates the table with initial employee data[cite: 354].

INSERT INTO employee (id, name, email, gender, designation) VALUES (1, 'Alice', 'alice@xyz.com', 'M', 'Software Developer'), (2, 'John', 'john@xyz.com', 'M', 'Software Manager'), (3, 'Ajay', 'ajay@xyz.com', 'M', 'Software Developer'), (4, 'Polina', 'polina@xyz.com', 'F', 'Sales Manager'), (5, 'Henry', 'henry@xyz.com', 'M', 'Senior Software Manager');

3. READ Data

Filters for 'Software Developer' employees[cite: 395].

SELECT * FROM employee WHERE designation = 'Software Developer';

4. UPDATE Record

Promotes Alice (ID 1) to Senior Developer[cite: 430].

UPDATE employee SET designation = 'Senior Software Developer' WHERE id = 1;

5. DELETE Record

Removes Ajay (ID 3) from the database.

DELETE FROM employee WHERE id = 3;
